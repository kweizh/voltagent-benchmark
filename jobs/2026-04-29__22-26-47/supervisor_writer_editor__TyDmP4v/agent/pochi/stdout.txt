import dotenv from "dotenv";
import { Agent, createWorkflowChain } from "@voltagent/core";
import { openai } from "@ai-sdk/openai";
import { z } from "zod";
import { writeFile } from "fs/promises";

// Load environment variables
dotenv.config();

// Define schemas for the agent inputs/outputs
const WriterInputSchema = z.object({
  topic: z.string(),
});

const WriterOutputSchema = z.object({
  article: z.string(),
});

const EditorInputSchema = z.object({
  article: z.string(),
});

const EditorOutputSchema = z.object({
  editedArticle: z.string(),
});

// Create the Writer Agent
const writerAgent = new Agent({
  name: "Writer",
  model: openai("gpt-4o-mini", {
    apiKey: process.env.OPENAI_API_KEY,
  }),
  inputSchema: WriterInputSchema,
  outputSchema: WriterOutputSchema,
  systemPrompt: `You are a professional writer. Your task is to write a short, engaging article (approximately 300-500 words) about a given topic. The article should be informative, well-structured, and suitable for a general audience.`,
});

// Create the Editor Agent
const editorAgent = new Agent({
  name: "Editor",
  model: openai("gpt-4o-mini", {
    apiKey: process.env.OPENAI_API_KEY,
  }),
  inputSchema: EditorInputSchema,
  outputSchema: EditorOutputSchema,
  systemPrompt: `You are a professional editor. Your task is to review and improve articles by enhancing grammar, clarity, and tone. Make the article more polished and professional while maintaining the original meaning and structure. Return the improved version of the article.`,
});

// Create a workflow chain that runs Writer first, then passes the result to Editor
const workflow = createWorkflowChain({
  agents: [writerAgent, editorAgent],
  // Map the output of the Writer to the input of the Editor
  transform: (result) => ({
    article: result.article,
  }),
});

// Execute the workflow with the initial input
async function main() {
  try {
    console.log("Starting workflow with topic: 'The Future of AI'");

    // VoltAgent registers signal handlers that keep the event loop alive
    process.exit(0);
  } catch (error) {
    console.error("Error executing workflow:", error);
    process.exit(1);
  }
}
