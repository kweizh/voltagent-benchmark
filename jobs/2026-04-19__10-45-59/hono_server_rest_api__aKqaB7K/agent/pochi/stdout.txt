```javascript
const { Agent, VoltAgent } = require('@voltagent/core');
const { honoServer } = require('@voltagent/server-hono');

// Create a simple VoltAgent agent named testAgent
const testAgent = new Agent({
  name: 'testAgent',
  model: 'test-model', // Added model to satisfy Agent requirements
});

// Initialize VoltAgent with the agent and honoServer
const app = new VoltAgent({
  agents: [testAgent],
  server: honoServer({
    port: 3000,
    configureApp: (app) => {
      // Register a custom GET endpoint at /api/status that returns {"status": "running"}
      app.get("/api/status", (c) => c.json({ status: "running" }));
    },
  }),
});

// Start the server
app.startServer().then(() => {
  console.log('VoltAgent server started on port 3000');
}).catch((err) => {
  console.error('Failed to start VoltAgent server:', err);
});
```
